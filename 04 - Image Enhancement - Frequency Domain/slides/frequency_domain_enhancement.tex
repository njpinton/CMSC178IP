\section{Introduction}

\begin{frame}{What is Frequency Domain Enhancement?}
\begin{alertblock}{Definition}
Frequency domain enhancement processes images by transforming to frequency domain, applying filters, and transforming back.
\end{alertblock}

\textbf{Mathematical Foundation:}
$$g(x,y) = \mathcal{F}^{-1}\{H(u,v) \cdot F(u,v)\}$$

Where: $f(x,y)$ = input, $g(x,y)$ = output, $H(u,v)$ = filter
\end{frame}

\begin{frame}{Why Use Frequency Domain?}
\begin{columns}
\column{0.5\textwidth}
\textbf{Advantages:}
\begin{itemize}
\item Direct frequency manipulation
\item Efficient periodic noise removal
\item FFT enables fast computation
\item Clear frequency visualization
\end{itemize}

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/frequency_domain_motivation.png}
\end{center}
\end{columns}
\end{frame}

\section{Step-by-Step Algorithm}

\begin{frame}{Frequency Domain Enhancement Algorithm}
\begin{algorithm}[H]
\SetKwInOut{Input}{Input}
\SetKwInOut{Output}{Output}
\Input{Original image $f(x,y)$ of size $M \times N$}
\Output{Enhanced image $g(x,y)$}
\BlankLine
\textbf{Step 1:} Pre-process image (zero-padding if needed)\;
\textbf{Step 2:} Center the transform: $f_c(x,y) = f(x,y) \cdot (-1)^{x+y}$\;
\textbf{Step 3:} Compute 2D FFT: $F(u,v) = \mathcal{F}\{f_c(x,y)\}$\;
\textbf{Step 4:} Design filter $H(u,v)$ based on enhancement goal\;
\textbf{Step 5:} Apply filter: $G(u,v) = H(u,v) \cdot F(u,v)$\;
\textbf{Step 6:} Compute inverse FFT: $g_c(x,y) = \mathcal{F}^{-1}\{G(u,v)\}$\;
\textbf{Step 7:} De-center: $g(x,y) = \text{Re}\{g_c(x,y) \cdot (-1)^{x+y}\}$\;
\textbf{Step 8:} Post-process (crop, normalize intensity)\;
\end{algorithm}
\end{frame}

\begin{frame}{Algorithm Implementation Considerations}
\textbf{Critical Implementation Details:}
\begin{itemize}
\item \textbf{Zero-padding:} Prevent wraparound artifacts
\item \textbf{Centering:} Move DC component to center for filter design
\item \textbf{Filter symmetry:} Ensure real output for real input
\item \textbf{Numerical precision:} Handle floating-point errors
\item \textbf{Memory management:} Optimize for large images
\end{itemize}

\vspace{0.5cm}
\textbf{Common Pitfalls:}
\begin{itemize}
\item Forgetting to center the spectrum
\item Not handling complex number precision
\item Improper filter normalization
\item Ignoring boundary effects
\end{itemize}
\end{frame}

\section{Core Theory}

\begin{frame}{2D Fourier Transform}
\begin{columns}
\column{0.5\textwidth}
\textbf{DFT Definition:}
$$F(u,v) = \frac{1}{MN} \sum_{x=0}^{M-1} \sum_{y=0}^{N-1} f(x,y) e^{-j2\pi(ux/M + vy/N)}$$

\textbf{Properties:}
\begin{itemize}
\item Complex coefficients
\item Periodic spectrum
\item DC at center
\end{itemize}

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/fourier_transform_2d.png}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{Frequency Representation}
\begin{columns}
\column{0.5\textwidth}
\textbf{Magnitude and Phase:}
$$|F(u,v)| = \sqrt{R^2(u,v) + I^2(u,v)}$$
$$\phi(u,v) = \arctan\left(\frac{I(u,v)}{R(u,v)}\right)$$

\textbf{Power Spectrum:}
$$P(u,v) = |F(u,v)|^2$$

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/frequency_representation.png}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{Computational Complexity Analysis}
\begin{columns}
\column{0.6\textwidth}
\textbf{Time Complexity Comparison:}
\begin{itemize}
\item \textbf{Direct DFT:} $O(M^2N^2)$ for $M \times N$ image
\item \textbf{FFT:} $O(MN \log(MN))$ - dramatic improvement!
\item \textbf{Spatial convolution:} $O(MNK^2)$ for $K \times K$ kernel
\item \textbf{Frequency multiplication:} $O(MN)$
\end{itemize}

\textbf{Memory Requirements:}
\begin{itemize}
\item Original image: $MN$ (real values)
\item Frequency domain: $2MN$ (complex values)
\item Working memory: $\sim 4MN$ total
\end{itemize}

\column{0.4\textwidth}
\textbf{Break-even Point:}

For kernel size $K$:
$$K > \sqrt{\log(MN)}$$

\textbf{Example:}

$512 \times 512$ image:
$K > 3.3$ pixels

FFT becomes efficient for kernels $\geq 4 \times 4$
\end{columns}
\end{frame}

\section{Filtering Methods}

\begin{frame}{Low-Pass Filtering}
\begin{columns}
\column{0.5\textwidth}
\textbf{Purpose:} Remove high frequencies (noise, details)

\textbf{Ideal Filter:}
$$H(u,v) = \begin{cases}
1 & \text{if } D(u,v) \leq D_0 \\
0 & \text{if } D(u,v) > D_0
\end{cases}$$

\textbf{Gaussian Filter:}
$$H(u,v) = e^{-D^2(u,v)/(2D_0^2)}$$

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/low_pass_filters.png}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{High-Pass Filtering}
\begin{columns}
\column{0.5\textwidth}
\textbf{Purpose:} Enhance edges and details

\textbf{High-pass Filter:}
$$H_{hp}(u,v) = 1 - H_{lp}(u,v)$$

\textbf{High-Frequency Emphasis:}
$$H_{hfe}(u,v) = a + b \cdot H_{hp}(u,v)$$

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/high_pass_filters.png}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{Band-Pass and Band-Reject Filtering}
\begin{columns}
\column{0.6\textwidth}
\textbf{Band-Pass Filter Design:}
$$H_{bp}(u,v) = \begin{cases}
1 & \text{if } D_1 \leq D(u,v) \leq D_2 \\
0 & \text{otherwise}
\end{cases}$$

\textbf{Band-Reject (Notch) Filter:}
$$H_{br}(u,v) = 1 - H_{bp}(u,v)$$

\textbf{Gaussian Band-Pass:}
$$H(u,v) = e^{-\frac{(D(u,v)-D_0)^2}{2W^2}}$$

Where $W$ is the bandwidth parameter.

\column{0.4\textwidth}
\textbf{Applications:}
\begin{itemize}
\item Remove periodic noise patterns
\item Enhance specific texture frequencies
\item Isolate signal frequencies
\item Remove scanner artifacts
\end{itemize}

\vspace{0.5cm}
\begin{center}
\includegraphics[width=\textwidth]{../figures/band_pass_filters.png}
\end{center}
\end{columns}
\end{frame}

\begin{frame}{Filter Comparison and Selection}
\begin{center}
\begin{tabular}{|l|c|c|c|c|}
\hline
\textbf{Filter Type} & \textbf{Sharpness} & \textbf{Ringing} & \textbf{Smoothness} & \textbf{Best Use} \\
\hline
Ideal & High & High & Low & Theoretical analysis \\
Butterworth & Medium & Medium & Medium & General purpose \\
Gaussian & Low & Low & High & Noise reduction \\
Chebyshev & High & Medium & Low & Sharp cutoffs \\
\hline
\end{tabular}
\end{center}

\vspace{0.5cm}
\textbf{Selection Guidelines:}
\begin{itemize}
\item \textbf{Gaussian:} When smoothness is critical (medical imaging)
\item \textbf{Butterworth:} Good balance for most applications
\item \textbf{Ideal:} When sharp frequency cutoff needed (notch filtering)
\item \textbf{Order selection:} Higher order = sharper transition, more ringing
\end{itemize}
\end{frame}

\begin{frame}{Homomorphic Filtering}
\begin{columns}
\column{0.5\textwidth}
\textbf{Purpose:} Separate illumination and reflectance

\textbf{Process:}
\begin{enumerate}
\item Take logarithm
\item Apply FFT
\item Filter with $H(u,v)$

\begin{frame}{Unsharp Masking in Frequency Domain}
\begin{columns}
\column{0.5\textwidth}
\textbf{Traditional Unsharp Masking:}
$$g(x,y) = f(x,y) + k \cdot [f(x,y) - f_{blur}(x,y)]$$

\textbf{Frequency Domain Equivalent:}
$$G(u,v) = F(u,v) \cdot [1 + k(1 - H_{lp}(u,v))]$$

\textbf{Advantages:}
\begin{itemize}
\item Controllable frequency-specific enhancement
\item No spatial artifacts
\item Efficient for large kernels
\item Precise frequency control
\end{itemize}

\column{0.5\textwidth}
\begin{center}
\includegraphics[width=\textwidth]{../figures/frequency_sharpening.png}
\end{center}

\textbf{Parameter Guidelines:}
\begin{itemize}
\item $k = 0.5-2.0$ for general enhancement
\item $D_0 = 0.1-0.3$ of image diagonal
\item Higher $k$ → stronger enhancement
\item Lower $D_0$ → sharper enhancement
\end{itemize}
\end{columns}
\end{frame}

\section{Implementation and Troubleshooting}

\begin{frame}{Practical Implementation Tips}
\textbf{Memory and Performance Optimization:}
\begin{itemize}
\item Use in-place FFT when possible (saves 50\% memory)
\item Process large images in overlapping blocks
\item Pre-compute filter masks for real-time applications
\item Use GPU acceleration for FFT computations
\end{itemize}

\vspace{0.5cm}
\textbf{Numerical Stability:}
\begin{itemize}
\item Handle zero-frequency components carefully
\item Use double precision for filter design
\item Normalize filters to prevent overflow
\item Check for NaN/Inf values after operations
\end{itemize}

\vspace{0.5cm}
\textbf{Quality Assessment:}
\begin{itemize}
\item Compare frequency spectra before/after
\item Monitor edge preservation metrics
\item Check for introduced artifacts
\item Validate enhancement with ground truth
\end{itemize}
\end{frame}

\begin{frame}{Common Problems and Solutions}
\begin{columns}
\column{0.5\textwidth}
\textbf{Problem:} Ringing artifacts
\begin{itemize}
\item \textbf{Cause:} Sharp filter transitions
\item \textbf{Solution:} Use Gaussian or Butterworth filters
\item \textbf{Prevention:} Gradual filter rolloff
\end{itemize}

\textbf{Problem:} Boundary effects
\begin{itemize}
\item \textbf{Cause:} Circular convolution assumption
\item \textbf{Solution:} Zero-padding or mirroring
\item \textbf{Prevention:} Proper image extension
\end{itemize}

\column{0.5\textwidth}
\textbf{Problem:} Loss of phase information
\begin{itemize}
\item \textbf{Cause:} Magnitude-only processing
\item \textbf{Solution:} Preserve or estimate phase
\item \textbf{Prevention:} Phase-aware filtering
\end{itemize}

\textbf{Problem:} Computational complexity
\begin{itemize}
\item \textbf{Cause:} Large image sizes
\item \textbf{Solution:} Block processing, GPU acceleration
\item \textbf{Prevention:} Algorithm selection based on size
\end{itemize}
\end{columns}
\end{frame}
